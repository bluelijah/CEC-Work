<script>
function cecTab(btn, idx) {
  var card = btn.closest('.cec-cp-card');
  card.querySelectorAll('.cec-cp-tab').forEach(function(t, i) {
    t.classList.toggle('active', i === idx);
    t.setAttribute('aria-selected', i === idx);
  });
  card.querySelectorAll('.cec-cp-panel').forEach(function(p, i) {
    p.classList.toggle('active', i === idx);
  });
}
</script>
