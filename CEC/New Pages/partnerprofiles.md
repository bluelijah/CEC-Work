<script>
var cecCpLast = null;
function cecCpOpen(name) {
  cecCpLast = document.activeElement;
  document.getElementById('cec-cp-otitle').textContent = name;
  document.getElementById('cec-cp-backdrop').classList.add('is-open');
  document.getElementById('cec-cp-closebtn').focus();
  document.body.style.overflow = 'hidden';
}
function cecCpClose() {
  document.getElementById('cec-cp-backdrop').classList.remove('is-open');
  document.body.style.overflow = '';
  if (cecCpLast) cecCpLast.focus();
}
document.getElementById('cec-cp-backdrop').addEventListener('click', function(e) {
  if (e.target === this) cecCpClose();
});
document.addEventListener('keydown', function(e) {
  if (e.key === 'Escape') cecCpClose();
});
</script>
